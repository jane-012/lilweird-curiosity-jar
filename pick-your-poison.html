create or replace function public.poison_draw(p_room uuid) returns void language plpgsql security definer set search_path='' as $$ declare r public.poison_rooms; chosen uuid; begin select * into r from public.poison_rooms where id = p_room for update; if not found or auth.uid() not in (r.host_id, r.guest_id) then raise exception 'Not a room member'; end if; if r.started_at is null or r.closed_at is not null or now() >= r.started_at + make_interval(secs => r.duration_seconds) then raise exception 'Room not active'; end if; select id into chosen from public.poison_room_cards where room_id = p_room and drawn_at is null order by random() limit 1; if chosen is null then raise exception 'No undrawn cards left'; end if; update public.poison_room_cards set drawn_at = now() where id = chosen; update public.poison_rooms set current_card = chosen, twist_visible = false where id = p_room; end; $$;
